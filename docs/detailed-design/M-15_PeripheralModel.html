1 功能概述

1.1 模块定位

M-15 外设模型层（Peripheral）实现外设抽象与注册表、实例化到地址区间；MVP 提供 UART、GPIO、Timer 三个内置外设（寄存器接口 + 行为级时序，决策点 2/ADR-2）。外设差异隔离在总线之下（3.2.5 表13）：M-14 按地址区间把访存分发给实例的 read/write；M-12 事件队列驱动 onEvent（定时器到期、输入就绪）；中断经 M-16（irqRequest/irqClear）；内部信号经 probe 上报观测管道（M-18）。

每个外设实例携带保真度分级元数据（寄存器级 / 行为级 / HDL 协同）供前端显式展示（评审 R5）；注册表插件化（NFR-06），新增外设类型不改既有实例。预留 port(name, dir)/link(peerPort) 端口抽象供 P1 多板互联（FR-24/25）。

1.3 MVP 范围与预留

MVP 实现范围：Peripheral 抽象接口（3.2.5 表13）、外设类型注册表（register/instantiate）、UART/GPIO/Timer 三个内置外设（寄存器接口 + 行为级时序）、实例化到地址区间（attach → M-14 mapRegion）、事件驱动回调（onEvent）、中断请求（irqRequest/irqClear）、探针上报（probe → M-18）、保真度分级元数据（fidelityMeta，评审 R5）、行为日志（log → M-21）。

2.1.1 复杂结构字段定义

PeripheralInstance 是实例化到地址区间的外设实体，字段定义如下：

// PeripheralInstance: 外设实例（attach 后挂在 instanceTable）
struct PeripheralInstance {
  char     type[16];         // 外设类型（UART/GPIO/TIMER/扩展注册类型）
  uint32_t baseAddr;         // 地址区间基址（M-14 mapRegion 登记）
  uint32_t size;             // 寄存器窗口大小（字节）
  FidelityMeta fidelity;     // 保真度分级元数据（评审 R5）
  void*    state;            // 外设内部状态（UartState/GpioState/TimerState）
  const void* config;        // 实例化配置（时钟/波特率/引脚数等）
  uint8_t  irqLine;          // 绑定的中断线（M-16，未用置 0xFF）
  // 行为层接口（3.2.5 表13）：
  int  (*read)(PeripheralInstance*, uint32_t offset, uint32_t* out);
  int  (*write)(PeripheralInstance*, uint32_t offset, uint32_t value);
  void (*onEvent)(PeripheralInstance*, uint64_t time, EventKind kind, void* payload);
  void (*probe)(PeripheralInstance*, const char* signal, uint32_t value, uint64_t time);
  void (*logMsg)(PeripheralInstance*, const char* msg);
  // 快照契约（评审 15-4）：外设内部状态（UART FIFO/Timer count/GPIO out）可序列化，
  // 纳入 M-20 L3 全状态快照/检查点重放（9.1），恢复后行为连续
  int (*snapshot)(PeripheralInstance*, void** out);
  int (*restore)(PeripheralInstance*, const void* snap);
};

fidelityMeta 描述该实例的建模深度（评审 R5）；三类配置结构逐字段定义如下：

// fidelityMeta: 保真度分级元数据（前端显式展示）
struct FidelityMeta {
  uint8_t level;             // 0=寄存器级 / 1=行为级（MVP 默认）/ 2=HDL 协同（P1）
  char    desc[32];          // 一句话说明（如 "行为级：波特率+CTS 流控"）
  uint8_t caps;              // 能力位：时序建模/中断/探针/流控 等
};
// UartCfg: UART 实例化配置（FR-07）
struct UartCfg { uint32_t clockHz; uint32_t baudRate;
                 uint8_t dataBits; uint8_t stopBits; uint8_t flowControl; }; // CTS/RTS
// UartState: UART 运行状态（评审 15-9 补 FIFO 深度与满时行为）
struct UartState {
  uint8_t txFifo[16];    // TX FIFO（默认深度 16 字节，估算基线）；满时写返回 E_PP-05 忙
  uint8_t rxFifo[16];    // RX FIFO（默认深度 16 字节）；满时输入注入丢弃并置 overrun 状态位（可选上报）
  uint16_t txHead, txTail, txBytes;   // TX 环形缓冲游标与待发字节数
  uint16_t rxHead, rxTail, rxBytes;   // RX 环形缓冲游标与已收字节数
  uint64_t bitAcc;       // 位时钟累加器（虚拟时间，ns）
  uint64_t lastTick;     // 上次位时钟推进时刻（虚拟时间）
  uint32_t bitIdx;       // 当前帧已发送位数（含 start/stop）
  uint8_t  txBusy;       // TX 忙标志
  uint8_t  cts;          // CTS 电平（流控）
  uint8_t  overrun;      // RX 溢出标志（FIFO 满时置位）
  const UartCfg* cfg;    // 实例化配置回指（baudRate/dataBits/stopBits/flowControl）
  PeripheralInstance* periph;  // 回指宿主实例（中断/探针用）
};
// GpioCfg: GPIO 实例化配置
struct GpioCfg { uint8_t pinCount; uint8_t direction; uint32_t initLevel; };
// TimerCfg: Timer 实例化配置
struct TimerCfg { uint32_t clockHz; uint8_t mode;
                  uint32_t prescaler; uint32_t compare; }; // compare 到期触发
// TimerState: Timer 运行状态
struct TimerState {
  uint64_t count;        // 计数值（按虚拟时间差快进，见 3.4）
  uint64_t lastEvent;    // 上次事件推进时刻（虚拟时间）
  uint64_t period;       // 到期间隔（虚拟时间，ns）= compare × prescaler 分频周期（评审 15-3）
  const TimerCfg* cfg;   // 实例化配置回指（clockHz/mode/prescaler/compare）
  PeripheralInstance* periph;  // 回指宿主实例（中断/探针用）
};

2.2 内置外设与保真度分级

外设寄存器接口（行为级窗口）行为时序探针信号（M-18）中断线
UART数据/状态/控制寄存器读写（收发 FIFO）波特率位时钟、CTS 流控（FR-07）、输入就绪事件TX（MVP 仅实现 TX 字节流；RX / CTS 未实现）rx-ready / tx-empty
GPIO引脚电平读写 + 方向控制引脚翻转OUT（GPIO 输出翻转，默认通道）—
Timer控制/预分频/比较寄存器比较值到期 → 事件入队 → 中断请求CMP（定时器比较，默认通道）timer-compare

保真度分级：寄存器级 = 仅状态寄存器读写（无时序）；行为级 = MVP 默认，在寄存器读写之上建模关键行为时序（波特率/CTS/比较触发）；HDL 协同 = P1，外接真实 HDL 仿真器。

2.3 数据流说明

下行：M-14 按地址区间分发访存 → 实例 read/write（寄存器窗口）；上行：行为时序产生事件 → M-12 onEvent 调度 / M-16 irqRequest；观测：probe 边沿信号 → M-18 → M-19 波形；记录：log 行为日志 → M-21。输入注入（UART 写/按键/GPIO 注入）经 M-04 recordInput 时间戳化入 M-20 事件流（4.7 表62）。

3 实现设计

3.1 Peripheral 抽象接口实现（3.2.5 表13）

外设差异隔离在总线之下：M-14 只按地址区间把访存交给实例 read/write；引擎只通过 onEvent 驱动；中断只经 irqRequest/irqClear 与 M-16 交互；观测只经 probe 上报。伪代码如下：

// read/write: 寄存器级访问入口（由 M-14 按地址区间分发到实例）
int periph_read(PeripheralInstance* p, uint32_t offset, uint32_t* out) {
    return p->read(p, offset, out);            // 转发到类型实现（UART/GPIO/Timer）
}
// attach: 实例化到地址区间（调用 M-14 mapRegion）
int periph_attach(PeripheralInstance* p, Bus* bus, uint32_t baseAddr) {
    Region r = { baseAddr, p->size, PERIPHERAL, p, 0 };
    int rc = bus->mapRegion(&r);               // 冲突/越界 → E_BUS-03/E_BUS-04
    if (rc == OK) { p->baseAddr = baseAddr; instanceTable[baseAddr] = p; }
    return rc;
}
// detach: 注销外设（评审 15-5）——unmap + instanceTable 清理 + 状态回收，
// 与 M-14 生命周期协议对齐（M-14 detach 强制清理对应区间，杜绝悬垂 target）
void periph_detach(PeripheralInstance* p, Bus* bus) {
    bus->unmap(p->baseAddr);                   // M-14 清理映射
    instanceTable.erase(p->baseAddr);          // 本模块清理登记
    release_state(p);                          // 状态回收（含 FIFO/计数缓冲释放）
}
// snapshot/restore: 外设状态快照（评审 15-4）——序列化 UART FIFO/Timer count/GPIO out，
// 纳入 M-20 L3 全状态快照/检查点重放；restore 前校验版本与 CRC（对齐 M-20 契约）
int periph_snapshot(PeripheralInstance* p, void** out)  { return p->snapshot(p, out); }
int periph_restore(PeripheralInstance* p, const void* s){ return p->restore(p, s); }
// onEvent: 事件驱动回调（由 M-12 事件队列驱动：定时器到期、输入就绪等）
void periph_onEvent(PeripheralInstance* p, uint64_t time, EventKind kind, void* payload) {
    p->onEvent(p, time, kind, payload);
}
// irqRequest / irqClear: 向中断控制器请求/清除中断线
void periph_irqRequest(PeripheralInstance* p) { intc_request(p->irqLine); }
void periph_irqClear(PeripheralInstance* p)    { intc_clear(p->irqLine); }
// probe: 向观测管道暴露内部信号（GPIO 翻转、定时器比较、中断脉冲）
void periph_probe(PeripheralInstance* p, const char* sig, uint32_t v, uint64_t t) {
    probe_sample(sig, v, t);                  // M-18 attachSignal 通道绑定
}
// precheck: 外设可访问性预检（评审 N1）——M-14 写拆分阶段一「预检所有片可访问后统一
//   执行」时调用；返回 err_t：OK 可访问 / E_PP-05 忙或内部不可访问；与 probe 探针上报
//   （void）语义分离，是独立预检接口（9.1.1 契约），不承载观测
int periph_precheck(PeripheralInstance* p, uint32_t offset) {
    if (!p || !p->state) return E_PP-05;          // 实例失效（未 attach/已 detach/悬垂）
    if (offset >= p->regWindow) return E_PP-05;   // 偏移超出寄存器窗口
    return p->precheck ? p->precheck(p, offset) : OK;  // 后端预检（忙/内部状态），缺省可访问
}
// log: 行为日志（NFR-07），用于 trace 与调试
void periph_log(PeripheralInstance* p, const char* msg) { log(INFO, "PERIPH", msg, ""); }
// port/link（P1 预留）：未连接端口产生告警（FR-25）

3.2 UART 实现

寄存器窗口（行为级）：数据寄存器（收发 FIFO 读写）、状态寄存器（TX 忙/RX 就绪/CTS 电平）、控制寄存器（使能/波特率/数据位/流控开关）。默认偏移布局作为实现基线（评审 15-8）：UART DATA=0x00 / STAT=0x04 / CTRL=0x08；GPIO DATA=0x00 / DIR=0x04；Timer CTRL=0x00 / PRE=0x04 / CMP=0x08——与具体固件驱动对齐作为变更项管理（变更走配置/文档变更流程，不静默改偏移）。行为时序：按 baudRate 生成位时钟（每 bit = 1e9/baudRate 虚拟时间），发送按 FIFO 逐字节推进，帧完成 = (1+dataBits+stopBits)×txBytes 位（含起始/停止位）；CTS 流控（FR-07）：CTS 无效时暂停发送；接收侧输入就绪触发 onEvent（输入注入经 M-20 事件流）或 rx-ready 中断。FIFO 深度（评审 15-9）：TX/RX 默认 16 字节；TX 满时写返回 E_PP-05 忙；RX 满时输入注入丢弃并置 overrun 状态位。运行中配置变更（评审 15-10）：写控制寄存器（使能/波特率/数据位）或 GPIO 方向寄存器时，位时钟/计数基准重置（bitAcc=0、lastTick=now），时序从新配置重新起算。

// UART 行为时序：位时钟推进（虚拟时间，评审 15-1 修正）
void uart_tick(UartState* s, uint64_t now) {
    if (s->cts && s->txBytes > 0) {           // CTS 有效且待发
        s->bitAcc += (now - s->lastTick);      // 按虚拟时间累加位时钟
        uint64_t bitT = (1ULL * 1e9) / s->cfg->baudRate;   // 每 bit 虚拟时间 = 1e9/baud（修正）
        while (s->bitAcc >= bitT) { s->bitAcc -= bitT; s->bitIdx++; }
        uint32_t bitsPerFrame = 1 + s->cfg->dataBits + s->cfg->stopBits;  // start + data + stop
        if (s->bitIdx >= bitsPerFrame * s->txBytes) {      // 帧完成判定计入 start/stop（修正）
            s->txBytes = 0; s->bitIdx = 0; s->txBusy = 0;
            periph_irqRequest(s->periph);                  // tx-empty 中断（FIFO 空）
        }
    }
    s->lastTick = now;
}

3.3 GPIO 实现

引脚电平读写 + 方向控制；引脚翻转时经 probe 上报边沿（M-18 默认通道 GPIO0），供波形视图（FR-10）。写入 GPIO 数据寄存器 → 电平变化 → probe("OUT", level, t_virt)。

// GPIO 写：电平变化即上报探针（边沿判定在 M-18）
int gpio_write(PeripheralInstance* p, uint32_t offset, uint32_t value) {
    GpioState* s = p->state;
    uint32_t changed = (s->out ^ value) & pinMask(p->config);
    s->out = value;
    if (changed) periph_probe(p, "OUT", value, virt_clock_now());  // 翻转探针（信号名取 SIGNAL_OUT）
    return OK;
}

3.4 Timer 实现

比较值触发：count 按虚拟时间差快进（评审 15-3）——每次到期处理时以（now − lastEvent）÷ 分频周期换算计数步进，达 compare 时经 M-12 scheduleEvent 入队到期事件，并 irqRequest 请求中断线（timer-compare）；处理器写控制寄存器清除。period（下次到期间隔）由配置推导：period = compare × 分频周期（分频周期 = 1e9/(clockHz/prescaler) ns），无需额外配置字段。职责边界（评审 N3）：定时中断权威源 = M-16 mtime/mtimecmp（标准 CLINT，固件标准路径经 M-14 MMIO）；本外设 timer-compare 中断线 MVP 不启用（避免双权威源计数不一致），Timer 外设仅作平台周期计数（count 可被固件读、比较探针供波形观测）；后续若启用，CMP 经 M-14 透传 mtimecmp（同一权威源），作为 P1 变更项管理。

// Timer 到期：count 快进（评审 15-3）+ 事件入队 + 中断请求
uint64_t prescaler_ticks(TimerState* s) {
    // 每计数步的虚拟时间（ns）= 1e9 / (clockHz / prescaler)
    return (1ULL * 1e9) / (s->cfg->clockHz / s->cfg->prescaler);
}
void timer_tick(TimerState* s, uint64_t now) {
    // count 推进：时间差 ÷ 分频周期（快进，不需逐周期事件）
    s->count += (now - s->lastEvent) / prescaler_ticks(s);
    if (s->count >= s->cfg->compare) {
        s->count = 0;
        s->period = s->cfg->compare * prescaler_ticks(s);   // 到期间隔（配置推导，无新增字段）
        engine_scheduleEvent(now + s->period, EV_TIMER, s); // M-12 事件队列
        if (s->timerIRQEnabled) intc_request(s->irqLine);   // 评审 N3：timer-compare 线 MVP 关闭
        periph_probe(s->periph, "CMP", 1, now);       // M-18 比较探针（观测保留；信号名取 SIGNAL_CMP）
    }
    s->lastEvent = now;
}
// 初始化：attach/使能时置 s->lastEvent = now；首个到期事件由首次 timer_tick 入队。

6.2 边界与降级

（1）探针溢出：按 M-19 溢出策略处置（默认丢最旧 + 标记，可配置暂停仿真），外设行为不因探针过载被中断（4.4.4.5、4.5.2.1）。

（2）采样可整体关闭：probe 为性能开关，关闭后外设行为不受影响（4.5.1.1）。

（3）配置校验失败保留旧配置：E_PP-02 拒绝新配置，既有实例状态不变（4.4.4.5）。

（4）确定性边界：外设事件必须入 M-12 事件队列稳定排序（虚拟时间, 插入序），不得依赖宿主时钟；违反 → 确定性断言失败（2.4 表6、源文档 7.3 表92）。

（5）中断风暴：由 M-16 按优先级 + 使能掩码抑制，未使能中断保持挂起不丢（4.4.5.5）；外设侧不额外兜底。

（6）外设运行时错误：read/write 返回 E_PP-05（忙/内部错误），总线层转 E_BUS-05 上抛（评审 15-6），与未映射 E_BUS-01 可区分；FIFO 满不静默吞写，行为留痕可查。
